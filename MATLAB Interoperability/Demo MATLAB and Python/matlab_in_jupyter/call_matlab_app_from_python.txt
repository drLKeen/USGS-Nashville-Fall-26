# Setup
import matlab.engine
m = matlab.engine.start_matlab()

 

# Access Data
from sklearn.datasets import load_boston
boston = load_boston()

boston.feature_names

 

#Preprocess Data
X_ml = matlab.double(boston.data.tolist())
y_ml = matlab.double(boston.target.tolist())

 

# AI modeling (Training using MATLAB Regression Learner)
m.regressionLearner(X_ml, y_ml, nargout = 0)

 

# AI modeling (Testing)
model = m.workspace['trainedModel']

predFcn = model.get('predictFcn')

 

X_test = X_ml[0]
